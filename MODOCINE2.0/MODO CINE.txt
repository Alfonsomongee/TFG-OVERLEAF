CineModePlayer.jsx	Reproductor principal. Gestiona scroll, detección de escena visible, cambios de acto, postMessage con Streamlit, y control de reproducción automática.
Scene.jsx	Envuelve cada bloque (ej. Bloque1KPI). Añade botón PDF y transiciones de entrada.
TimelineController.jsx	Barra de control con slider, botones de reproducción, velocidad, skip intro, spotlight y modo kiosco.
chartRegistry.js	Mapeo de nombres de escenas a componentes React lazy, más metadatos (título, acto, descripción).
ActTransition.jsx	Animación de fundido a negro cuando se cambia de acto.
ParticleSystem.jsx	Genera partículas visuales según el acto (polvo solar, glitch, ceniza, líneas fluidas).
PDFExportButton.jsx	Captura el contenido de la escena y genera un PDF con html2canvas + jsPDF.
Bloque*.jsx	Componentes de visualización (KPI, mix, cascada, frecuencia, interconexiones, cronología). Todos aceptan el tema forense y ya tienen su propio CSS.

export default function Scene({ scene, isActive, onEnter, onLeave, currentTime }) {
  const sceneRef = useRef(null);
  const Component = chartRegistry[scene.component];

  useEffect(() => {
    const observer = new IntersectionObserver(
      (entries) => {
        entries.forEach((entry) => {
          if (entry.isIntersecting) {
            onEnter?.(scene.id);
            entry.target.classList.add('cine-scene');
          } else {
            onLeave?.(scene.id);
          }
        });
      },
      { threshold: 0.3 }
    );

    if (sceneRef.current) {
      observer.observe(sceneRef.current);
    }

    return () => {
      if (sceneRef.current) observer.unobserve(sceneRef.current);
    };
  }, [scene.id, onEnter, onLeave]);

  return (
    <div 
      ref={sceneRef} 
      className="cine-scene-wrapper"
      data-scene-id={scene.id}
      style={{ opacity: 0, marginBottom: '4rem' }}
    >
      <div className="cine-act-header">
        <div className="cine-act-title">ACTO {scene.act}</div>
        <div className="cine-act-subtitle">{scene.title}</div>
      </div>
      <Suspense fallback={<LoadingPlaceholder />}>
        <Component />
      </Suspense>
    </div>
  );
}

import React, { useState, useEffect, useRef } from 'react';

export default function TimelineController({ scenes, currentSceneId, onSceneChange, isPlaying, onPlayPause }) {
  const [progress, setProgress] = useState(0);
  const currentIndex = scenes.findIndex(s => s.id === currentSceneId);
  
  useEffect(() => {
    setProgress(((currentIndex + 1) / scenes.length) * 100);
  }, [currentIndex, scenes.length]);

  const handleSliderChange = (e) => {
    const index = Math.floor((parseFloat(e.target.value) / 100) * scenes.length);
    const targetScene = scenes[Math.min(index, scenes.length - 1)];
    if (targetScene) onSceneChange(targetScene.id);
  };

  return (
    <div className="timeline-controller">
      <div className="timeline-controls">
        <button className="timeline-btn" onClick={onPlayPause}>
          {isPlaying ? '⏸' : '▶'}
        </button>
      </div>
      
      <input
        type="range"
        className="timeline-slider"
        min="0"
        max="100"
        step="1"
        value={progress}
        onChange={handleSliderChange}
      />
      
      <div className="timeline-progress-text">
        {currentIndex + 1} / {scenes.length}
      </div>
    </div>
  );
}

export default function CineModePlayer() {
  const [currentSceneId, setCurrentSceneId] = useState(sceneMetadata[0]?.id || '');
  const [isPlaying, setIsPlaying] = useState(false);
  const [autoScrollEnabled, setAutoScrollEnabled] = useState(true);
  const sceneRefs = useRef({});
  const containerRef = useRef(null);

  // Sincronizar con Streamlit (postMessage)
  useEffect(() => {
    const handleMessage = (event) => {
      if (event.data.type === 'SYNC_SCENE') {
        const targetScene = sceneMetadata.find(s => s.id === event.data.sceneId);
        if (targetScene && sceneRefs.current[targetScene.id]) {
          sceneRefs.current[targetScene.id].scrollIntoView({ behavior: 'smooth' });
          setCurrentSceneId(targetScene.id);
        }
      }
    };
    window.addEventListener('message', handleMessage);
    
    // Notificar que el reproductor está listo
    window.parent.postMessage({ type: 'CINE_MODE_READY' }, '*');
    
    return () => window.removeEventListener('message', handleMessage);
  }, []);

  // Detectar escena visible por scroll
  useEffect(() => {
    if (!autoScrollEnabled) return;
    
    const observer = new IntersectionObserver(
      (entries) => {
        const visibleScenes = entries.filter(e => e.isIntersecting);
        if (visibleScenes.length > 0) {
          const topScene = visibleScenes.reduce((prev, curr) => 
            curr.boundingClientRect.top < prev.boundingClientRect.top ? curr : prev
          );
          const sceneId = topScene.target.getAttribute('data-scene-id');
          if (sceneId && sceneId !== currentSceneId) {
            setCurrentSceneId(sceneId);
            // Notificar a Streamlit
            window.parent.postMessage({ type: 'SCENE_CHANGE', sceneId }, '*');
          }
        }
      },
      { threshold: 0.3 }
    );
    
    Object.values(sceneRefs.current).forEach(ref => {
      if (ref) observer.observe(ref);
    });
    
    return () => observer.disconnect();
  }, [currentSceneId, autoScrollEnabled]);

  const handleSceneEnter = useCallback((sceneId) => {
    console.log(`📽️ Escena activa: ${sceneId}`);
    // Aquí puedes añadir lógica para activar audio ambiental, etc.
  }, []);

  const handleSceneChange = (sceneId) => {
    if (sceneRefs.current[sceneId]) {
      sceneRefs.current[sceneId].scrollIntoView({ behavior: 'smooth' });
      setCurrentSceneId(sceneId);
      setAutoScrollEnabled(true);
    }
  };

  const togglePlayPause = () => {
    setIsPlaying(!isPlaying);
    if (!isPlaying) {
      // Reproducción automática: desplazamiento suave cada X segundos
      let index = sceneMetadata.findIndex(s => s.id === currentSceneId);
      const interval = setInterval(() => {
        if (!isPlaying) return;
        if (index < sceneMetadata.length - 1) {
          index++;
          handleSceneChange(sceneMetadata[index].id);
        } else {
          clearInterval(interval);
          setIsPlaying(false);
        }
      }, 8000);
      return () => clearInterval(interval);
    }
  };

  return (
    <div className="cine-mode-container" ref={containerRef}>
      {sceneMetadata.map((scene, idx) => (
        <div
          key={scene.id}
          ref={(el) => { sceneRefs.current[scene.id] = el; }}
          data-scene-id={scene.id}
          style={{ scrollMarginTop: '80px' }}
        >
          <Scene
            scene={scene}
            isActive={currentSceneId === scene.id}
            onEnter={handleSceneEnter}
            currentTime={null}
          />
        </div>
      ))}
      
      <TimelineController
        scenes={sceneMetadata}
        currentSceneId={currentSceneId}
        onSceneChange={handleSceneChange}
        isPlaying={isPlaying}
        onPlayPause={togglePlayPause}
      />
    </div>
  );
}

  const animationRef = useRef(null);
  const particlesRef = useRef([]);
  const [dimensions, setDimensions] = useState({ width: 0, height: 0 });
  const config = PARTICLE_CONFIGS[act] || PARTICLE_CONFIGS[1];

  // Inicializar partículas
  const initParticles = (width, height) => {
    const particles = [];
    for (let i = 0; i < config.count; i++) {
      particles.push({
        x: Math.random() * width,
        y: Math.random() * height,
        size: config.size * (0.5 + Math.random() * 1.5),
        speedX: (Math.random() - 0.5) * config.speed,
        speedY: (Math.random() - 0.5) * config.speed,
        color: config.color,
        shape: config.shape,
        life: Math.random() * 100,
        glitchOffset: config.glitch ? Math.random() * 10 : 0
      });
    }
    particlesRef.current = particles;
  };

  // Actualizar dimensiones
  useEffect(() => {
    const updateDimensions = () => {
      setDimensions({
        width: window.innerWidth,
        height: window.innerHeight
      });

    updateDimensions();
    window.addEventListener('resize', updateDimensions);
    return () => window.removeEventListener('resize', updateDimensions);
  }, []);

    if (dimensions.width && dimensions.height) {
      initParticles(dimensions.width, dimensions.height);
    }
  }, [dimensions, config]);

  useEffect(() => {
    const canvas = canvasRef.current;
    if (!canvas) return;
    const ctx = canvas.getContext('2d');
    if (!ctx) return;

    const animate = () => {
      if (!canvas || !ctx) return;
      ctx.clearRect(0, 0, dimensions.width, dimensions.height);

        // Actualizar posición

        p.y += p.speedY;
        
        // Rebote en bordes (o reaparecer)
        if (p.x < 0) p.x = dimensions.width;
        if (p.x > dimensions.width) p.x = 0;
        if (p.y < 0) p.y = dimensions.height;
        if (p.y > dimensions.height) p.y = 0;

        let drawY = p.y;

          drawY += (Math.random() - 0.5) * p.glitchOffset;
          opacityMod = 0.8 + Math.random() * 0.5;
        }
        
        ctx.save();
        if (p.shape === 'square') {
          ctx.fillStyle = `rgba(${p.color[0]}, ${p.color[1]}, ${p.color[2]}, ${config.opacity * opacityMod})`;

          ctx.strokeStyle = `rgba(${p.color[0]}, ${p.color[1]}, ${p.color[2]}, ${config.opacity * opacityMod})`;

          ctx.fill();
        }
        ctx.restore();
      });
      
      animationRef.current = requestAnimationFrame(animate);
    };
    
    animate();
    return () => {
      if (animationRef.current) cancelAnimationFrame(animationRef.current);
    };
  }, [dimensions, config]);

  return <canvas ref={canvasRef} className="particle-canvas" width={dimensions.width} height={dimensions.height} />;

export default function ParticleBackground({ act }) {
  const canvasRef = useRef(null);
  const animationRef = useRef(null);
  const particlesRef = useRef([]);

  useEffect(() => {
    const canvas = canvasRef.current;
    if (!canvas) return;
    const ctx = canvas.getContext('2d');
    let width = window.innerWidth;
    let height = window.innerHeight;
    
    const config = PARTICLE_CONFIGS[act] || PARTICLE_CONFIGS[1];
    
    // Inicializar partículas
    const initParticles = () => {
      const particles = [];
      for (let i = 0; i < config.count; i++) {
        particles.push({
          x: Math.random() * width,
          y: Math.random() * height,
          size: config.size.min + Math.random() * (config.size.max - config.size.min),
          speedX: (Math.random() - 0.5) * config.speed,
          speedY: (Math.random() - 0.5) * config.speed,
          opacity: Math.random() * config.opacity,
          flicker: config.flicker ? Math.random() * 0.1 : 0
        });
      }
      particlesRef.current = particles;
    };
    
    const drawParticle = (p) => {
      ctx.save();
      ctx.globalAlpha = p.opacity + (p.flicker ? Math.sin(Date.now() * 0.005 * p.flicker) * 0.1 : 0);
      ctx.fillStyle = config.color;
      
      if (config.shape === 'square') {
        ctx.fillRect(p.x - p.size/2, p.y - p.size/2, p.size, p.size);
      } else if (config.shape === 'line') {
        ctx.beginPath();
        ctx.moveTo(p.x - p.size, p.y);
        ctx.lineTo(p.x + p.size, p.y);
        ctx.strokeStyle = config.color;
        ctx.lineWidth = 1;
        ctx.stroke();
      } else {
        ctx.beginPath();
        ctx.arc(p.x, p.y, p.size, 0, Math.PI * 2);
        ctx.fill();
      }
      ctx.restore();
    };
    
    const updateParticle = (p) => {
      if (config.fallDirection === 'down') {
        p.y += p.speedY;
        if (p.y > height) p.y = 0;
      } else if (config.direction === 'up') {
        p.y -= p.speedY;
        if (p.y < 0) p.y = height;
      } else {

        p.y += p.speedY;
        if (p.x < 0) p.x = width;

        if (p.y < 0) p.y = height;
        if (p.y > height) p.y = 0;

      if (!ctx) return;

      canvas.height = height;

    };

          if (p.x > canvas.width) p.x = 0;
          if (p.y < 0) p.y = canvas.height;
          if (p.y > canvas.height) p.y = 0;
        }
        
        ctx.save();
        ctx.globalAlpha = p.alpha;
        ctx.fillStyle = config.color;
        ctx.strokeStyle = config.color;
        
        if (config.shape === 'square') {
          ctx.fillRect(p.x, p.y, p.size, p.size);
        } else if (config.shape === 'line') {
          ctx.beginPath();
          ctx.moveTo(p.x, p.y);
          ctx.lineTo(p.x + p.length, p.y + p.length * 0.5);
          ctx.stroke();
        } else {
          ctx.beginPath();
          ctx.arc(p.x, p.y, p.size, 0, Math.PI * 2);
          ctx.fill();
        }
        ctx.restore();
      });
      
      animationRef.current = requestAnimationFrame(animate);
    };
    
    animationRef.current = requestAnimationFrame(animate);
    
    return () => {
      if (animationRef.current) cancelAnimationFrame(animationRef.current);
      window.removeEventListener('resize', resize);
    };
  }, [act, config]);

  return (
    <canvas
      ref={canvasRef}
      style={{
        position: 'fixed',
        top: 0,
        left: 0,
        width: '100%',
        height: '100%',
        pointerEvents: 'none',
        zIndex: 1
      }}
    />
  );
}

Este modo oculta los controles y la interfaz de navegación, dejando solo el contenido de las escenas en pantalla completa, ideal para ferias, exposiciones o demostraciones autónomas.

// src/components/cine-mode/CineModePlayer.jsx
// Añadir estado:
const [kioskMode, setKioskMode] = useState(false);

// Función para alternar modo kiosco
const toggleKioskMode = () => {
  if (!kioskMode) {
    // Entrar en modo kiosco
    document.body.classList.add('kiosk-mode');
    if (document.fullscreenEnabled) {
      document.documentElement.requestFullscreen();
    }
  } else {
    // Salir del modo kiosco
    document.body.classList.remove('kiosk-mode');
    if (document.fullscreenEnabled && document.fullscreenElement) {
      document.exitFullscreen();
    }
  }
  setKioskMode(!kioskMode);
};

// Efecto para manejar el cambio de escena automático en modo kiosco
useEffect(() => {
  if (kioskMode) {
    // Reproducción automática lenta (opcional)
    const interval = setInterval(() => {
      const currentIdx = sceneMetadata.findIndex(s => s.id === currentSceneId);
      if (currentIdx < sceneMetadata.length - 1) {
        const nextScene = sceneMetadata[currentIdx + 1];
        const nextElement = sceneRefs.current[nextScene.id];
        if (nextElement) {
          nextElement.scrollIntoView({ behavior: 'smooth' });
          setCurrentSceneId(nextScene.id);
        }
      } else {
        // Volver al inicio
        const firstScene = sceneMetadata[0];
        if (firstScene && sceneRefs.current[firstScene.id]) {
          sceneRefs.current[firstScene.id].scrollIntoView({ behavior: 'smooth' });
          setCurrentSceneId(firstScene.id);
        }
      }
    }, 15000); // Avanza cada 15 segundos
    return () => clearInterval(interval);
  }
}, [kioskMode, currentSceneId, sceneMetadata]);

📄 Mejora 9: Exportar informe PDF de la escena actual
Vamos a implementar la generación de un PDF con el contenido de la escena activa (título, descripción, gráfica visible). Esto permite al tribunal o a los usuarios llevarse un registro impreso del análisis.

1. Instalar dependencias necesarias

npm install html2canvas jspdf
2. Crear componente PDFExportButton.jsx
jsx
// src/components/cine-mode/PDFExportButton.jsx
import React, { useState } from 'react';
import html2canvas from 'html2canvas';
import jsPDF from 'jspdf';

export default function PDFExportButton({ targetRef, title }) {
  const [isExporting, setIsExporting] = useState(false);

  const handleExport = async () => {
    if (!targetRef?.current) return;
    setIsExporting(true);
    
    try {
      const canvas = await html2canvas(targetRef.current, {
        scale: 2,
        backgroundColor: '#0a0a0c',
        logging: false,
        useCORS: true
      });
      
      const imgData = canvas.toDataURL('image/png');
      const pdf = new jsPDF('p', 'mm', 'a4');
      const imgWidth = 190; // mm
      const pageHeight = 277; // mm
      const imgHeight = (canvas.height * imgWidth) / canvas.width;
      let position = 10;
      
      pdf.addImage(imgData, 'PNG', 10, position, imgWidth, imgHeight);
      
      // Añadir título y metadata
      pdf.setFontSize(16);
      pdf.setTextColor(40, 40, 40);
      pdf.text(title || 'Análisis forense', 10, 10);
      pdf.setFontSize(10);
      pdf.text(`Generado: ${new Date().toLocaleString()}`, 10, pageHeight - 10);
      
      pdf.save(`forense_${title?.replace(/\s/g, '_') || 'escena'}.pdf`);
    } catch (error) {
      console.error('Error generando PDF:', error);
      alert('Error al generar el PDF. Inténtalo de nuevo.');
    } finally {
      setIsExporting(false);
    }
  };

  return (
    <button 
      className="timeline-btn pdf-export-btn" 
      onClick={handleExport}
      disabled={isExporting}
      title="Exportar esta escena a PDF"
    >
      {isExporting ? '⏳ Generando...' : '📄 PDF'}
    </button>
  );
}
