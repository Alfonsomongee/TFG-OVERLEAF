```jsx

// SwingEquationSimulator.jsx



import React, { useState, useEffect, useRef } from 'react';

import { motion, AnimatePresence } from 'framer-motion';

import { LineChart, Line, XAxis, YAxis, CartesianGrid, Tooltip, Legend, ResponsiveContainer, ReferenceLine, ComposedChart, BarChart, Bar } from 'recharts';

import { useBaseUrl } from '@docusaurus/useBaseUrl';

import GlitchTitle from '../GlitchTitle/GlitchTitle';

import styles from './SwingEquationSimulator.module.css';



const SwingEquationSimulator = () => {

&#x20; // ============ State: Slider Parameters ============

&#x20; const \[H, setH] = useState(2.45); // Inertia constant (seconds)

&#x20; const \[deltaP, setDeltaP] = useState(8.8); // Power imbalance (% of load)

&#x20; const \[D, setD] = useState(1.5); // Damping factor (pu)

&#x20; const \[syntheticInertia, setSyntheticInertia] = useState(0); // FFR gain

&#x20; const \[weakGrid, setWeakGrid] = useState(false); // SCR < 2.0 simulation

&#x20; const \[uflsStages, setUflsStages] = useState(6); // Number of active UFLS stages



&#x20; // ============ State: Simulation Control ============

&#x20; const \[isRunning, setIsRunning] = useState(false);

&#x20; const \[simulationTime, setSimulationTime] = useState(0); // seconds

&#x20; const \[frequency, setFrequency] = useState(50.0); // Hz

&#x20; const \[deltaF, setDeltaF] = useState(0); // Per-unit frequency deviation

&#x20; const \[rocof, setRocof] = useState(0); // dF/dt (Hz/s)

&#x20; const \[historyData, setHistoryData] = useState(\[]); // For chart

&#x20; const \[currentUflsStage, setCurrentUflsStage] = useState(0); // Which UFLS stage is active

&#x20; const \[systemStatus, setSystemStatus] = useState('NORMAL');



&#x20; // ============ Refs ============

&#x20; const simulationIntervalRef = useRef(null);

&#x20; const freqHistoryRef = useRef(\[]);

&#x20; const rocofHistoryRef = useRef(\[]);

&#x20; const previousFreqRef = useRef(50.0);



&#x20; // ============ Load configuration JSON ============

&#x20; const \[params, setParams] = useState(null);



&#x20; useEffect(() => {

&#x20;   fetch(useBaseUrl('/data/swing\_equation\_params.json'))

&#x20;     .then((res) => res.json())

&#x20;     .then((data) => setParams(data))

&#x20;     .catch((err) => console.error('Error loading swing equation params:', err));

&#x20; }, \[]);



&#x20; // ============ Simulation Engine: Euler Integration ============

&#x20; const runSimulationStep = (currentH, currentDeltaP, currentD, currentSynth, currentWeakGrid, time) => {

&#x20;   const DT\_MS = 10; // 10ms step

&#x20;   const DT\_S = DT\_MS / 1000;

&#x20;   const F0 = 50.0;

&#x20;   const OMEGA\_S = 314.159; // 2π \* 50 Hz



&#x20;   // Get current frequency from refs (for derivative)

&#x20;   const prevFreq = previousFreqRef.current;



&#x20;   // Euler update: 2H \* d(Δf)/dt + D\*Δf = ΔP

&#x20;   // => d(Δf)/dt = (ΔP - D\*Δf) / (2\*H)

&#x20;   const dDeltaF\_dt = (currentDeltaP - D \* deltaF) / (2 \* currentH);



&#x20;   // Add synthetic inertia response (proportional to df/dt)

&#x20;   // Synthetic inertia injects power = -FFR\_gain \* df/dt

&#x20;   // This reduces ΔP

&#x20;   let effectiveDeltaP = currentDeltaP;

&#x20;   if (syntheticInertia > 0) {

&#x20;     effectiveDeltaP -= syntheticInertia \* dDeltaF\_dt;



&#x20;     // Weak grid oscillation: if weak\_grid is TRUE and synthetic inertia is high,

&#x20;     // introduce subsynchronous feedback (0.63 Hz oscillation)

&#x20;     if (currentWeakGrid \&\& syntheticInertia > 2.0) {

&#x20;       const oscillationAmplitude = 0.5 \* syntheticInertia; // Grows with FFR gain

&#x20;       const oscillationFreq = 0.63; // Hz, matching Granada pre-blackout oscillation

&#x20;       const phaseShift = 2 \* Math.PI \* oscillationFreq \* time;

&#x20;       effectiveDeltaP += oscillationAmplitude \* Math.sin(phaseShift);

&#x20;     }

&#x20;   }



&#x20;   // Recalculate with effective ΔP

&#x20;   const dDeltaF\_dt\_effective = (effectiveDeltaP - D \* deltaF) / (2 \* currentH);



&#x20;   // Euler step

&#x20;   const newDeltaF = deltaF + dDeltaF\_dt\_effective \* DT\_S;

&#x20;   const newFreq = F0 \* (1 + newDeltaF);

&#x20;   const newRocof = (newFreq - prevFreq) / DT\_S;



&#x20;   previousFreqRef.current = newFreq;



&#x20;   // ============ UFLS Logic ============

&#x20;   // Check UFLS thresholds and update effectiveDeltaP accordingly

&#x20;   let cumulativeLoadShed = 0;

&#x20;   const uflsThresholds = \[

&#x20;     { freq: 49.0, shed: 15, stage: 1 },

&#x20;     { freq: 48.7, shed: 15, stage: 2 },

&#x20;     { freq: 48.4, shed: 10, stage: 3 },

&#x20;     { freq: 48.0, shed: 10, stage: 4 },

&#x20;   ];



&#x20;   let activeStage = 0;

&#x20;   for (let i = 0; i < Math.min(uflsStages, uflsThresholds.length); i++) {

&#x20;     if (newFreq <= uflsThresholds\[i].freq) {

&#x20;       activeStage = i + 1;

&#x20;       cumulativeLoadShed = uflsThresholds.slice(0, i + 1).reduce((sum, u) => sum + u.shed, 0);

&#x20;     }

&#x20;   }



&#x20;   setCurrentUflsStage(activeStage);



&#x20;   // Update status badge

&#x20;   let status = 'NORMAL';

&#x20;   if (newFreq > 49.8) status = 'NORMAL';

&#x20;   else if (newFreq > 49.0) status = 'UFLS ACTIVE';

&#x20;   else if (newFreq > 48.0) status = 'CRITICAL';

&#x20;   else status = 'COLLAPSE';



&#x20;   setSystemStatus(status);



&#x20;   // Update state

&#x20;   setDeltaF(newDeltaF);

&#x20;   setFrequency(newFreq);

&#x20;   setRocof(newRocof);



&#x20;   // Append to history for charting

&#x20;   const dataPoint = {

&#x20;     time: time.toFixed(2),

&#x20;     t: time,

&#x20;     frequency: parseFloat(newFreq.toFixed(2)),

&#x20;     rocof: parseFloat(newRocof.toFixed(3)),

&#x20;     deltaF: parseFloat(newDeltaF.toFixed(4)),

&#x20;     loadShed: cumulativeLoadShed,

&#x20;   };



&#x20;   freqHistoryRef.current.push(dataPoint);



&#x20;   // Keep only last 300 points for performance

&#x20;   if (freqHistoryRef.current.length > 300) {

&#x20;     freqHistoryRef.current.shift();

&#x20;   }



&#x20;   setHistoryData(\[...freqHistoryRef.current]);

&#x20; };



&#x20; // ============ Simulation Loop ============

&#x20; useEffect(() => {
```

```mdx
\# Simulador Interactivo: La Ecuación del Swing



\## Introducción Teórica



La \*\*Ecuación del Swing\*\* es el modelo fundamental de la estabilidad transitoria de sistemas eléctricos. Describe cómo la frecuencia de la red cambia en respuesta a un desbalance entre potencia suministrada (P\_m) y potencia demandada (P\_e).



\### La Forma Linealizada (Dominio de Frecuencia)



Para un simulador educativo en tiempo real, usamos la forma linealizada de la ecuación del Swing:



$$

2H \\frac{d(\\Delta f)}{dt} + D \\Delta f = \\Delta P

$$



\*\*Dónde:\*\*

\- $H$ = Constante de inercia equivalente del sistema (segundos)

\- $\\Delta f$ = Desviación de frecuencia en pu: $\\Delta f = (f - 50) / 50$

\- $D$ = Factor de amortiguamiento (pu)

\- $\\Delta P$ = Desbalance de potencia activa (% de la carga)



\### Interpretación Física



\- \*\*$H$ (inercia)\*\* determina la \*\*rapidez inicial\*\* de cambio de frecuencia (ROCOF). Menor H = ROCOF más alto = colapso más rápido.

\- \*\*$D$ (amortiguamiento)\*\* determina el \*\*estado estacionario\*\* final. Mayor D = menor caída de frecuencia en estado permanente (pero solo si las reservas primarias pueden responder).



\*\*Punto crítico:\*\* La inercia no previene el colapso; solo lo retrasa. Sin potencia activa disponible (ΔP negativo y sostenido), el sistema colapsará independientemente de H.

\## Parámetros Explicados



\### Inercia del Sistema (H)



El valor de $H$ en segundos representa cuánto tiempo podría una máquina síncrona suministrar su potencia nominal extrayendo solo su energía cinética almacenada.



\*\*Valores típicos:\*\*



| Tecnología | H (segundos) |

|---|---|

| Generador nuclear grande | 8–9 |

| Turbina de gas (CCGT) | 6–7 |

| Hidroeléctrica | 3–4 |

| Turbina eólica (Tipo 4, converter completo) | 0 |

| Solar PV | 0 |

| Condensador síncrono | 1–2 |



\*\*En el 28 de abril de 2025:\*\*

\- Sistema ibérico con 78% renovables: \*\*H = 2.45 segundos\*\* (crítico)

\- Sistema ibérico histórico (2015, más carbón/nuclear): \*\*H ≈ 5 segundos\*\* (seguro)



\### Pérdida de Potencia (ΔP)



Representa el porcentaje de carga total que se pierde repentinamente. En el 28A, la pérdida de 2,200 MW en un sistema de 25,000 MW = \*\*8.8%\*\*.



Esta es la "magnitud del disturbio". Mayor ΔP = mayor ROCOF = colapso más rápido.



\### Amortiguamiento (D)



Representa dos efectos:

1\. \*\*Carga dependiente de frecuencia:\*\* Los motores de inducción consumen menos potencia a menor frecuencia, proporcionando un "alivio automático"

2\. \*\*Respuesta del regulador de voltaje (AVR):\*\* Las máquinas síncronas ajustan automáticamente su salida ante cambios de frecuencia



Valores típicos: \*\*D = 1.5 pu\*\* (baseline)



\### Inercia Sintética (FFR - Fast Frequency Response)



Los inversores modernos pueden programarse para inyectar potencia activa en proporción a la tasa de cambio de frecuencia:



$$

P\_{FFR} = -K\_{FFR} \\frac{df}{dt}

$$



Esto imita artificialmente el comportamiento inercial de una máquina síncrona. \*\*Pero con una caverna importante:\*\* En redes débiles (bajo SCR), las altas ganancias FFR causan inestabilidad de control en lugar de ayudar.



\### Red Débil (SCR < 2.0)



\*\*Short Circuit Ratio (SCR)\*\* mide la "fortaleza" de la red:



$$

\\text{SCR} = \\frac{S\_{cc}}{P\_n}

$$



\- \*\*SCR > 3:\*\* Red fuerte (voltaje estable)

\- \*\*2 < SCR < 3:\*\* Red media

\- \*\*SCR < 2:\*\* Red débil (voltaje inestable)

\- \*\*SCR < 1.5:\*\* Red muy débil (inestabilidad del PLL)



En el sur de España (Granada, Badajoz), el SCR era < 2.0 debido a la altísima concentración de solar sin soporte síncrono.



\---



\## Las Tres Interacciones Pedagógicas Clave



\### 1️⃣ \*\*El Retraso de la Inercia\*\*



\*\*Acción:\*\* Carga "Condiciones 28A" y observa el ROCOF ≈ 0.9 Hz/s. Luego carga "+ Inercia" (H=4.5s) y observa el mismo evento.



\*\*Observación:\*\* El ROCOF se reduce a ≈ 0.6 Hz/s, pero la frecuencia sigue cayendo y alcanza aproximadamente el mismo valor mínimo.



\*\*Lección:\*\* \*\*La inercia no es una batería.\*\* No previene el colapso; solo retrasa cuándo ocurre, comprando tiempo para que las reservas primarias respondan. Sin potencia disponible, el colapso es inevitable.



\---



\### 2️⃣ \*\*La Trampa de la Inercia Sintética en Redes Débiles\*\*



\*\*Acción:\*\* Carga "FFR Sintética" con "Red Débil" habilitada. Verás que el ROCOF inicial se reduce gracias a la inyección FFR. Pero...



\*\*Observación:\*\* La frecuencia comienza a \*\*oscilar violentamente\*\* a \~0.63 Hz, el rango subsincrónico. El sistema se vuelve más inestable, no menos.



\*\*Lección:\*\* \*\*El control del inversor tiene límites.\*\* En redes débiles, los PLL no pueden rastrear correctamente el voltaje cuando el propio inversor está inyectando potencia. Esto crea un bucle de realimentación que causa la misma instabilidad de 0.63 Hz observada en Granada antes del apagón.



\---



\### 3️⃣ \*\*Los Límites del UFLS\*\*



\*\*Acción:\*\* Ejecuta "Condiciones 28A". Verás que UFLS se activa en etapas (49.0 Hz, luego 48.7 Hz, etc.) reduciendo la carga.



\*\*Observación:\*\* Cuando ROCOF excede 1.0 Hz/s y la frecuencia cae por debajo de 48.0 Hz, incluso el UFLS máximo no detiene el colapso.



\*\*Lección:\*\* \*\*UFLS es la última línea de defensa, pero tiene un punto de quiebre.\*\* Una vez que la inercia es tan baja que ROCOF > 1.0 Hz/s, las máquinas síncronas se desenganchan por "pole-slip" (pérdida de sincronismo). En ese punto, el UFLS está jugando un juego perdido.



\---



\## Ecuación del Swing: Forma Completa (Referencia)



Para los estudiosos, aquí está la forma no lineal completa:



$$

\\frac{d^2 \\delta}{dt^2} = \\frac{\\omega\_s}{2H}(P\_m - P\_e(\\delta) - D\\frac{d\\delta}{dt})

$$



Donde $P\_e(\\delta) = \\frac{EV}{X}\\sin(\\delta)$ es la potencia eléctrica en función del ángulo del rotor $\\delta$.



El simulador utiliza una versión linealizada que omite el ángulo y se enfoca directamente en la frecuencia, que es lo pedagógicamente más importante.
```

\## Technical Notes



1\. \*\*Integration step:\*\* 10ms (0.01s) is critical for stability with H as low as 1.0s

2\. \*\*ROCOF calculation:\*\* Derived as (f\[n] - f\[n-1]) / dt, updated every 10ms step

3\. \*\*Weak grid oscillation:\*\* When SCR toggle is TRUE and synthetic\_inertia > 2.0, add a 0.63 Hz sinusoidal perturbation to ΔP

4\. \*\*UFLS paradox:\*\* While the UFLS logic is implemented correctly (reduces demand), the real-world Ferranti effect (overvoltage from unloaded transmission lines) is not modeled. This is noted in the educational context but is a simplification for browser-based pedagogy.

5\. \*\*Synthetic inertia formula:\*\* $P\_{FFR} = -K\_{FFR} \\times df/dt$ (current approach: reduce effective ΔP by this amount)

6\. \*\*Chart performance:\*\* Keep only last 300 data points in memory to avoid React re-render lag
