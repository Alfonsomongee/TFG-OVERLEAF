🔧 1. Proxy serverless mejorado api/esios-multi.js
Este proxy obtiene todos los indicadores de una sola vez (mucho más eficiente).

javascript
// api/esios-multi.js
export default async function handler(req, res) {
  const token = process.env.ESIOS_TOKEN || '59f41cbd500501a872390d7e3d838b29ea20b2e55b3fa96153adf270840b11f5';
  
  // Lista completa de indicadores que usaremos
  const indicators = {
    demanda: 1293,
    solar: 2044,
    eolica: 2038,
    nuclear: 2039,
    hidro: 2042,
    gas: 2041,
    renovable_total: 10351,
    no_renovable_total: 10352,
    intercambio_neto: 2043,
    export_francia: 2071,
    export_portugal: 2070,
    precio_spot: 600,
  };
  
  const results = {};
  
  for (const [key, id] of Object.entries(indicators)) {
    try {
      const url = `https://api.esios.ree.es/indicators/${id}`;
      const response = await fetch(url, {
        headers: { 'Accept': 'application/json', 'x-api-key': token }
      });
      const data = await response.json();
      const lastValue = data.indicator.values[data.indicator.values.length - 1]?.value;
      results[key] = lastValue ?? null;
    } catch (err) {
      results[key] = null;
    }
  }
  
  // Añadir timestamp
  results.timestamp = new Date().toISOString();
  
  res.status(200).json(results);
}

import React, { useState, useEffect } from 'react';
import Plot from 'react-plotly.js';

const BLACKOUT = {
  solar: 19155, eolica: 3540, nuclear: 3870, gas: 990, hidro: 2000
};

export default function MixGeneracion() {
  const [data, setData] = useState(null);
  const [loading, setLoading] = useState(true);

  const fetchData = async () => {
    try {
      const res = await fetch('/api/esios-multi');
      const json = await res.json();
      setData(json);
    } catch (err) {
      console.error(err);
    } finally {
      setLoading(false);
    }
  };

  useEffect(() => {
    fetchData();
    const interval = setInterval(fetchData, 300000);
    return () => clearInterval(interval);
  }, []);

  if (loading || !data) return <div>Cargando mix de generación...</div>;

  const hoy = {
    Solar: data.solar || 0,
    Eólica: data.eolica || 0,
    Nuclear: data.nuclear || 0,
    Gas: data.gas || 0,
    Hidráulica: data.hidro > 0 ? data.hidro : 0,
  };
  const hoyPenetracion = (data.renovable_total / (data.renovable_total + data.no_renovable_total)) * 100;
  const blackoutPenetracion = 84.5;
  const delta = hoyPenetracion - blackoutPenetracion;

  const donutHoy = { labels: Object.keys(hoy), values: Object.values(hoy) };
  const donut28A = { labels: Object.keys(BLACKOUT), values: Object.values(BLACKOUT) };

  return (
    <div style={{ padding: '1rem' }}>
      <h2>🍩 Mix de generación: Hoy vs 28-A</h2>
      <div style={{ display: 'flex', gap: '2rem', flexWrap: 'wrap', justifyContent: 'center' }}>
        <div>
          <h3>Hoy {new Date(data.timestamp).toLocaleTimeString()}</h3>
          <Plot data={[{
            values: donutHoy.values, labels: donutHoy.labels, type: 'pie', hole: 0.4,
            marker: { colors: ['#f1c40f', '#2ecc71', '#e74c3c', '#95a5a6', '#3498db'] }
          }]} layout={{ width: 400, height: 400, showlegend: true }} />
        </div>
        <div>
          <h3>28 de abril de 2025 (colapso)</h3>
          <Plot data={[{
            values: donut28A.values, labels: donut28A.labels, type: 'pie', hole: 0.4,
            marker: { colors: ['#f1c40f', '#2ecc71', '#e74c3c', '#95a5a6', '#3498db'] }
          }]} layout={{ width: 400, height: 400, showlegend: false }} />
        </div>
      </div>
      <div style={{ textAlign: 'center', marginTop: '1rem', fontWeight: 'bold' }}>
        Penetración renovable: {hoyPenetracion.toFixed(1)}% (28-A: 84.5%) 
        <span style={{ color: delta > 0 ? '#e74c3c' : '#2ecc71' }}>
          {delta > 0 ? ` 🔴 +${delta.toFixed(1)}%` : ` 🟢 ${delta.toFixed(1)}%`}
        </span>
      </div>
    </div>
  );
}

import React, { useState, useEffect } from 'react';
import Plot from 'react-plotly.js';

export default function DemandaRenovableTrend() {
  const [current, setCurrent] = useState(null);
  const [historical, setHistorical] = useState(null);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    fetch('/api/esios-multi')
      .then(res => res.json())
      .then(setCurrent)
      .catch(console.error);
    fetch('/data/blackout_snapshot_28A.json')
      .then(res => res.json())
      .then(setHistorical)
      .finally(() => setLoading(false));
  }, []);

  useEffect(() => {
    const interval = setInterval(() => {
      fetch('/api/esios-multi').then(setCurrent).catch(console.error);
    }, 300000);
    return () => clearInterval(interval);
  }, []);

  if (loading || !current || !historical) return <div>Cargando...</div>;

  const data = [
    { variable: 'Demanda (MW)', hoy: current.demanda, historical: historical.demanda },
    { variable: 'Renovable total (MW)', hoy: current.renovable_total, historical: historical.renovable_total },
  ];

  return (
    <div>
      <h2>📈 Demanda y generación renovable: Hoy vs 28-A</h2>
      <Plot data={[
        { x: data.map(d => d.variable), y: data.map(d => d.hoy), type: 'bar', name: 'Hoy', marker: { color: '#06b6d4' } },
        { x: data.map(d => d.variable), y: data.map(d => d.historical), type: 'bar', name: '28-A', marker: { color: '#ef4444' } }
      ]} layout={{ title: 'Comparativa', barmode: 'group', yaxis: { title: 'MW' } }} />
    </div>
  );
}

import React, { useState, useEffect } from 'react';

  const [loading, setLoading] = useState(true);

  useEffect(() => {
    fetch('/api/esios-multi')
      .then(res => res.json())
      .then(setData)
      .finally(() => setLoading(false));

          type: 'indicator', mode: 'gauge+number', value: penetracion,
          gauge: { axis: { range: [0, 100] }, bar: { color: penetracion > 75 ? '#ef4444' : '#10b981' },
                   steps: [{ range: [0, 50], color: '#10b981' }, { range: [50, 75], color: '#f59e0b' }, { range: [75, 100], color: '#ef4444' }] }

      </div>
      <div>

6️⃣ Radar de vulnerabilidad (RadarVulnerabilidad.jsx)
jsx
import React, { useState, useEffect } from 'react';
import Plot from 'react-plotly.js';

export default function RadarVulnerabilidad() {
  const [data, setData] = useState(null);
  const [historical, setHistorical] = useState(null);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    Promise.all([
      fetch('/api/esios-multi').then(res => res.json()),
      fetch('/data/blackout_snapshot_28A.json').then(res => res.json())
    ]).then(([current, hist]) => {
      setData(current);
      setHistorical(hist);
      setLoading(false);
    }).catch(console.error);

    const interval = setInterval(() => {
      fetch('/api/esios-multi').then(setData).catch(console.error);
    }, 300000);
    return () => clearInterval(interval);
  }, []);

  if (loading || !data || !historical) return <div>Cargando...</div>;

  const categories = ['Penetración renovable', 'Inercia estimada', 'Exportación neta', 'Demanda'];
  const hoyValues = [
    (data.renovable_total / (data.renovable_total + data.no_renovable_total)) * 100,
    (data.nuclear * 5 + (data.hidro > 0 ? data.hidro * 3 : 0) + data.gas * 4) / 1000,
    -data.intercambio_neto / 1000,
    data.demanda / 1000,
  ];
  const historicalValues = [
    historical.penetracion_renovable,
    historical.inercia_estimada,
    -historical.intercambio_neto / 1000,
    historical.demanda / 1000,
  ];

  return (
    <div>
      <h2>📊 Radar de vulnerabilidad</h2>
      <Plot data={[
        { r: hoyValues, theta: categories, type: 'scatterpolar', name: 'Hoy', fill: 'toself', marker: { color: '#06b6d4' } },
        { r: historicalValues, theta: categories, type: 'scatterpolar', name: '28-A (colapso)', fill: 'toself', marker: { color: '#ef4444' } }
      ]} layout={{ polar: { radialaxis: { range: [0, 100] } }, title: 'Distancia al perfil de colapso' }} />
      <p>Valores más altos = mayor riesgo. Cuanto más se aproxime el polígono azul al rojo, más vulnerable es el sistema.</p>
    </div>
  );
}
