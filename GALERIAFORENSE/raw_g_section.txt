📊 Propuesta de visualización
Gráfica 1: Evolución semanal del almacenamiento hidroeléctrico (Área / Línea)
Propósito: Mostrar la variación estacional de los embalses y detectar anomalías. Se aprecia claramente un mínimo en primavera (semanas 1-4) y un máximo a mediados de año (semanas 20-25). El apagón ocurrió en la semana 18 (finales de abril), que coincide con un punto de almacenamiento intermedio-alto (~15.000 GWh).

Código React + Recharts:

jsx
import React from 'react';
import {
  AreaChart,
  Area,
  XAxis,
  YAxis,
  CartesianGrid,
  Tooltip,
  ResponsiveContainer,
  ReferenceLine,
  Label
} from 'recharts';

const data = [ /* tu JSON aquí */ ];

// Encontrar la semana 18 (apagón)
const blackoutWeek = data.find(d => d.week === 18);

const formatMWh = (value) => {
  if (value >= 1e6) return `${(value / 1e6).toFixed(1)} TWh`;
  if (value >= 1e3) return `${(value / 1e3).toFixed(0)} GWh`;
  return `${value} MWh`;
};

export default function HydroReservoirChart() {
  return (
    <ResponsiveContainer width="100%" height={400}>
      <AreaChart data={data} margin={{ top: 20, right: 30, left: 20, bottom: 20 }}>
        <CartesianGrid strokeDasharray="3 3" opacity={0.3} />
        <XAxis
          dataKey="week"
          label={{ value: 'Semana del año', position: 'insideBottom', offset: -5 }}
          tickFormatter={(w) => `S${w}`}
        />
        <YAxis
          tickFormatter={formatMWh}
          label={{ value: 'Almacenamiento (MWh)', angle: -90, position: 'insideLeft' }}
          domain={[0, 16000000]}
        />
        <Tooltip
          formatter={(value) => formatMWh(value)}
          labelFormatter={(w) => `Semana ${w}`}
          contentStyle={{ backgroundColor: 'var(--ifm-background-surface-color)', border: '1px solid var(--ifm-color-emphasis-200)' }}
        />
        <Area
          type="monotone"
          dataKey="mwh"
          stroke="#1abc9c"
          fill="#1abc9c"
          fillOpacity={0.3}
          strokeWidth={2}
        />
        {blackoutWeek && (
          <ReferenceLine
            x={blackoutWeek.week}
            stroke="#ef4444"
            strokeWidth={2}
            strokeDasharray="6 6"
            label={
              <Label
                value="⚡ Apagón 28-A"
                position="top"
                fill="#ef4444"
                fontSize={12}
                fontWeight="bold"
              />
            }
          />
        )}
      </AreaChart>
    </ResponsiveContainer>
  );
}
