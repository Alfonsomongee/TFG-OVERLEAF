import json
import pandas as pd
import plotly.graph_objects as go
from typing import Optional

def generar_grafica_voltajes(
    json_path: str = "data_28a_voltage_profiles.json",
    slider_time: Optional[str] = None
) -> go.Figure:
    """
    Genera la gráfica de perfiles de tensión (28-A).
    slider_time: timestamp (str) para añadir línea vertical sincronizada.
    """
    with open(json_path, "r") as f:
        data = json.load(f)

    timestamps = pd.to_datetime(data["timestamps"])
    voltages = data["voltages"]

    fig = go.Figure()

    for nudo, valores in voltages.items():
        fig.add_trace(go.Scatter(
            x=timestamps,
            y=valores,
            mode="lines",
            name=nudo,
            line=dict(width=2)
        ))

    # Bandas de tensión
    fig.add_hrect(y0=0.95, y1=1.05, fillcolor="rgba(0,255,0,0.1)", line_width=0, annotation_text="Nominal ±5%")
    fig.add_hrect(y0=1.05, y1=1.10, fillcolor="rgba(255,165,0,0.2)", line_width=0, annotation_text="Alerta")
    fig.add_hrect(y0=1.10, y1=1.20, fillcolor="rgba(255,0,0,0.3)", line_width=0, annotation_text="Crítico")

    # Línea del disparo
    fig.add_vline(x=pd.Timestamp("2025-04-28 12:32:57"), line_dash="dash", line_color="red",
                  annotation_text="Disparo Granada", annotation_position="top right")

    # Línea vertical del slider (si existe)
    if slider_time:
        try:
            slider_ts = pd.Timestamp(slider_time)
            fig.add_vline(x=slider_ts, line_dash="dot", line_color="#06b6d4", line_width=1.5,
                          annotation_text="Cursor", annotation_position="bottom")
        except:
            pass

    fig.update_layout(
        title="Perfiles de tensión por nudo crítico (28-A)",
        xaxis_title="Hora",
        yaxis_title="Tensión (p.u.)",
        template="plotly_dark",
        hovermode="x unified"
    )
    return fig

import plotly.graph_objects as go
import pandas as pd
from typing import Optional

def generar_timeline(slider_time: Optional[str] = None) -> go.Figure:
    """
    Genera timeline de eventos del 28-A.
    slider_time: timestamp para línea vertical sincronizada.
    """
    eventos = [
        ("12:03", "Primera oscilación forzada", "yellow"),
        ("12:19", "Segunda oscilación", "yellow"),
        ("12:32:57", "Disparo transformer Granada (-355 MW)", "orange"),
        ("12:33:16", "Desconexiones masivas inversores", "red"),
        ("12:33:20", "Apertura Marruecos", "red"),
        ("12:33:21", "Pérdida sincronismo Francia", "red"),
        ("12:33:27", "COLAPSO TOTAL", "black"),
    ]

    # Alturas alternadas para evitar solapamiento
    alturas = [-40, -80, -120, -40, -80, -120, -40]

    fig = go.Figure()

    for i, (hora, desc, color) in enumerate(eventos):
        fig.add_annotation(
            x=hora,
            y=1,
            text=desc,
            showarrow=True,
            arrowhead=2,
            arrowcolor=color,
            ax=0,
            ay=alturas[i % len(alturas)],
            font=dict(color="white", size=11),
            bgcolor="rgba(0,0,0,0.6)",
            bordercolor=color,
            borderwidth=1
        )

    # Línea base horizontal
    fig.add_hline(y=1, line_dash="dot", line_color="gray", annotation_text="Línea de eventos")

    # Línea vertical del slider
    if slider_time:
        try:
            slider_ts = pd.Timestamp(slider_time)
            fig.add_vline(x=slider_ts.strftime("%H:%M:%S"), line_dash="dot", line_color="#06b6d4",
                          annotation_text="Cursor", annotation_position="bottom")
        except:
            pass

    fig.update_layout(
        title="Eventos críticos del 28-A",
        xaxis_title="Hora",
        yaxis_title="",
        xaxis=dict(range=["11:55", "12:40"], tickformat="%H:%M:%S"),
        template="plotly_dark",
        height=300,
        showlegend=False
    )
    return fig

def calcular_voltaje_stress(voltage_pu: Optional[float]) -> float:
    """
    Si voltage_pu es None o no existe, asume 1.0 (tensión nominal) -> estrés 0.
    """
    if voltage_pu is None:
        return 0.0
    # Solo consideramos sobretensión (peligro para el 28-A)
    if voltage_pu <= 1.05:
        return 0.0
    stress = (voltage_pu - 1.05) / 0.10 * 100
    return min(100, max(0, stress))
