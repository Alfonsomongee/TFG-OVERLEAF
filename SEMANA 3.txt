```javascript

// audioSonification.js



export class GridSonifier {

&#x20; constructor() {

&#x20;   this.audioContext = null;

&#x20;   this.oscillatorPrimary = null;

&#x20;   this.oscillatorStress = null;

&#x20;   this.filterStress = null;

&#x20;   this.gainMaster = null;

&#x20;   this.lfoFrequency = null;

&#x20;   this.lfoVoltage = null;

&#x20;   this.isActive = false;

&#x20; }



&#x20; // Initialize Web Audio API (requires user gesture)

&#x20; initialize() {

&#x20;   if (this.audioContext) return; // Already initialized



&#x20;   const AudioContext = window.AudioContext || window.webkitAudioContext;

&#x20;   if (!AudioContext) {

&#x20;     console.warn('Web Audio API not supported in this browser');

&#x20;     return false;

&#x20;   }



&#x20;   this.audioContext = new AudioContext();



&#x20;   // Master gain (volume)

&#x20;   this.gainMaster = this.audioContext.createGain();

&#x20;   this.gainMaster.gain.value = 0.3; // Conservative volume

&#x20;   this.gainMaster.connect(this.audioContext.destination);



&#x20;   // Primary oscillator: grid frequency (50 Hz → 47 Hz during collapse)

&#x20;   this.oscillatorPrimary = this.audioContext.createOscillator();

&#x20;   this.oscillatorPrimary.type = 'sine';

&#x20;   this.oscillatorPrimary.frequency.value = 50; // Start at nominal



&#x20;   // Stress oscillator: voltage harmonics

&#x20;   this.oscillatorStress = this.audioContext.createOscillator();

&#x20;   this.oscillatorStress.type = 'sawtooth'; // Rich harmonics

&#x20;   this.oscillatorStress.frequency.value = 50;



&#x20;   // Stress filter: cutoff controlled by voltage level

&#x20;   this.filterStress = this.audioContext.createBiquadFilter();

&#x20;   this.filterStress.type = 'lowpass';

&#x20;   this.filterStress.frequency.value = 50; // Silent at low voltage

&#x20;   this.filterStress.Q.value = 2;



&#x20;   // LFO for frequency modulation (0.63 Hz oscillation)

&#x20;   this.lfoFrequency = this.audioContext.createOscillator();

&#x20;   this.lfoFrequency.frequency.value = 0.63; // Hz

&#x20;   const lfoGainFreq = this.audioContext.createGain();

&#x20;   lfoGainFreq.gain.value = 0.5; // Modulation depth



&#x20;   // LFO for voltage modulation

&#x20;   this.lfoVoltage = this.audioContext.createOscillator();

&#x20;   this.lfoVoltage.frequency.value = 0.21; // Hz

&#x20;   const lfoGainVolt = this.audioContext.createGain();

&#x20;   lfoGainVolt.gain.value = 0.3;



&#x20;   // Routing

&#x20;   this.lfoFrequency.connect(lfoGainFreq);

&#x20;   lfoGainFreq.connect(this.oscillatorPrimary.frequency);



&#x20;   this.lfoVoltage.connect(lfoGainVolt);

&#x20;   lfoGainVolt.connect(this.gainMaster.gain);



&#x20;   this.oscillatorPrimary.connect(this.gainMaster);

&#x20;   this.oscillatorStress.connect(this.filterStress);

&#x20;   this.filterStress.connect(this.gainMaster);



&#x20;   // Start oscillators

&#x20;   const now = this.audioContext.currentTime;

&#x20;   this.oscillatorPrimary.start(now);

&#x20;   this.oscillatorStress.start(now);

&#x20;   this.lfoFrequency.start(now);

&#x20;   this.lfoVoltage.start(now);



&#x20;   this.isActive = true;

&#x20;   return true;

&#x20; }



&#x20; // Update frequency (Hz)

&#x20; setGridFrequency(frequencyHz) {

&#x20;   if (!this.oscillatorPrimary) return;

&#x20;   this.oscillatorPrimary.frequency.setTargetAtTime(

&#x20;     frequencyHz,

&#x20;     this.audioContext.currentTime,

&#x20;     0.05 // Time constant for exponential ramp (50ms)

&#x20;   );

&#x20; }



&#x20; // Update voltage stress level (pu, normalized 0-1.1)

&#x20; setVoltageStress(voltagepu) {

&#x20;   if (!this.filterStress) return;



&#x20;   // Map voltage (pu) to filter cutoff frequency

&#x20;   // At V ≤ 1.0 pu: cutoff = 50 Hz (stressharmonics silent)

&#x20;   // At V = 1.1 pu: cutoff = 5000 Hz (harsh stressharmonics)

&#x20;   const normalizedV = Math.max(1.0, Math.min(1.1, voltagepu));

1. &#x20;   const vRange = normalizedV - 1.0; // 0 to 0.1

&#x20;   const cutoffHz = 50 + (vRange / 0.1) \* 4950; // 50 Hz → 5000 Hz



&#x20;   this.filterStress.frequency.setTargetAtTime(

&#x20;     cutoffHz,

&#x20;     this.audioContext.currentTime,

&#x20;     0.05

&#x20;   );

&#x20; }



&#x20; // Stop audio

&#x20; stop() {

&#x20;   if (!this.isActive) return;



&#x20;   const now = this.audioContext.currentTime;

&#x20;   this.oscillatorPrimary?.stop(now + 0.1);

&#x20;   this.oscillatorStress?.stop(now + 0.1);

&#x20;   this.lfoFrequency?.stop(now + 0.1);

&#x20;   this.lfoVoltage?.stop(now + 0.1);



&#x20;   this.oscillatorPrimary = null;

&#x20;   this.oscillatorStress = null;

&#x20;   this.lfoFrequency = null;

&#x20;   this.lfoVoltage = null;

&#x20;   this.isActive = false;

&#x20; }



&#x20; // Get audio context state

&#x20; getState() {

&#x20;   return {

&#x20;     isActive: this.isActive,

&#x20;     contextState: this.audioContext?.state || 'offline',

&#x20;   };

&#x20; }

}
```
